function greet(name) {
  const hour = new Date().getHours();
  const time = hour < 12 ? "Buenos días"
             : hour < 18 ? "Buenas tardes"
             :                "Buenas noches";
  return `${time}, ${name}!`;
}

console.log(greet("mundo"));
// → "Buenas noches, mundo!"
